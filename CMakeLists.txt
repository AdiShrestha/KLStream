cmake_minimum_required(VERSION 3.20)
project(KLStream VERSION 0.3.1 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(klstream INTERFACE)
add_library(KLStream::klstream ALIAS klstream)
target_include_directories(klstream INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/source/include)
target_compile_features(klstream INTERFACE cxx_std_17)
target_link_libraries(klstream INTERFACE Threads::Threads)
option(KLSTREAM_BUILD_TESTS "Build offline foundation correctness tests" ON)
option(KLSTREAM_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(KLSTREAM_BUILD_TESTS)
  enable_testing()
  add_executable(engine_tests source/tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE KLStream::klstream)
  target_compile_options(engine_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(KLSTREAM_SANITIZERS)
    target_compile_options(engine_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(engine_tests PRIVATE -fsanitize=address,undefined)
  endif()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_executable(forest_reference_dump source/tests/forest_reference_dump.cpp)
    target_link_libraries(forest_reference_dump PRIVATE KLStream::klstream)
    target_compile_options(forest_reference_dump PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(KLSTREAM_SANITIZERS)
      target_compile_options(forest_reference_dump PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(forest_reference_dump PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME forest_independent_reference COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/source/tests/check_forest_reference.py $<TARGET_FILE:forest_reference_dump>)
    set_tests_properties(forest_independent_reference PROPERTIES TIMEOUT 30)
  else()
    message(STATUS "Python3 unavailable: independent forest reference check not executed")
  endif()
  add_executable(engine_runner source/experiments/engine_runner.cpp)
  target_link_libraries(engine_runner PRIVATE KLStream::klstream)
  target_compile_options(engine_runner PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(KLSTREAM_SANITIZERS)
    target_compile_options(engine_runner PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(engine_runner PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME engine_runner_help COMMAND engine_runner --help)
  set_tests_properties(engine_runner_help PROPERTIES TIMEOUT 15)
  if(Python3_Interpreter_FOUND)
    add_test(NAME experiment_runner_test COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/source/tests/test_runner.py)
    set_tests_properties(experiment_runner_test PROPERTIES TIMEOUT 45)
  endif()

  add_test(NAME engine_correctness COMMAND engine_tests)
  set_tests_properties(engine_correctness PROPERTIES TIMEOUT 45)
endif()
