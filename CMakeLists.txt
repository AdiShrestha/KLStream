cmake_minimum_required(VERSION 3.20)
project(KLStream VERSION 0.3.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(klstream INTERFACE)
add_library(KLStream::klstream ALIAS klstream)
target_include_directories(klstream INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/source/include)
target_compile_features(klstream INTERFACE cxx_std_17)
target_link_libraries(klstream INTERFACE Threads::Threads)
option(KLSTREAM_BUILD_TESTS "Build offline foundation correctness tests" ON)
option(KLSTREAM_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(KLSTREAM_BUILD_TESTS)
  enable_testing()
  add_executable(engine_tests source/tests/engine_tests.cpp)
  target_link_libraries(engine_tests PRIVATE KLStream::klstream)
  target_compile_options(engine_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(KLSTREAM_SANITIZERS)
    target_compile_options(engine_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(engine_tests PRIVATE -fsanitize=address,undefined)
  endif()
  add_test(NAME engine_correctness COMMAND engine_tests)
  set_tests_properties(engine_correctness PROPERTIES TIMEOUT 45)
endif()
